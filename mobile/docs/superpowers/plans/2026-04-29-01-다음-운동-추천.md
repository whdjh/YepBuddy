// src/entities/workout-session/lib/weeklyPlan.ts
import type { BodyPart, StoredWorkoutSession } from "../model/types"

const ROTATION: BodyPart[] = ["chest", "back", "legs", "shoulders", "arms", "core"]

export function getNextRecommendedBodyPart(
  last: StoredWorkoutSession | null
): BodyPart {
  if (!last || last.bodyParts.length === 0) return ROTATION[0]
  const dominant = last.bodyParts.reduce((max, bp) =>
    bp.setCount > max.setCount ? bp : max
  )
  const idx = ROTATION.indexOf(dominant.part)
  return ROTATION[idx === -1 ? 0 : (idx + 1) % ROTATION.length]
}
```

// src/features/start-workout/model/useNextWorkoutSuggestion.ts
import { useEffect, useState } from "react"
import {
  getLatestStoredWorkoutSession,
  getNextRecommendedBodyPart,
} from "@/entities/workout-session"
import type { BodyPart } from "@/entities/workout-session"

export function useNextWorkoutSuggestion(): {
  suggestedPart: BodyPart
  isLoading: boolean
} {
  const [suggestedPart, setSuggestedPart] = useState<BodyPart>("chest")
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    getLatestStoredWorkoutSession()
      .then((last) => setSuggestedPart(getNextRecommendedBodyPart(last)))
      .finally(() => setIsLoading(false))
  }, [])

  return { suggestedPart, isLoading }
}
```
