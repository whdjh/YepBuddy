```typescript
import { workoutReducer, initialWorkoutState } from "../workoutState"
import type { WorkoutState } from "../workoutState"

const stateWithChest: WorkoutState = {
  ...initialWorkoutState,
  bodyParts: [{ part: "chest", setCount: 3 }],
}

test("adds a detail to an existing part", () => {
  const next = workoutReducer(stateWithChest, {
    type: "TOGGLE_BODY_PART_DETAIL",
    payload: { part: "chest", detail: "upper_chest" },
  })
  expect(next.bodyParts[0].details).toEqual(["upper_chest"])
})

test("removes a detail that is already set", () => {
  const state: WorkoutState = {
    ...initialWorkoutState,
    bodyParts: [{ part: "chest", setCount: 3, details: ["upper_chest"] }],
  }
  const next = workoutReducer(state, {
    type: "TOGGLE_BODY_PART_DETAIL",
    payload: { part: "chest", detail: "upper_chest" },
  })
  expect(next.bodyParts[0].details).toEqual([])
})

test("does nothing if part is not selected", () => {
  const next = workoutReducer(initialWorkoutState, {
    type: "TOGGLE_BODY_PART_DETAIL",
    payload: { part: "chest", detail: "upper_chest" },
  })
  expect(next.bodyParts).toEqual([])
})

test("appends without removing existing details", () => {
  const state: WorkoutState = {
    ...initialWorkoutState,
    bodyParts: [{ part: "chest", setCount: 3, details: ["upper_chest"] }],
  }
  const next = workoutReducer(state, {
    type: "TOGGLE_BODY_PART_DETAIL",
    payload: { part: "chest", detail: "lower_chest" },
  })
  expect(next.bodyParts[0].details).toEqual(["upper_chest", "lower_chest"])
})
```

```typescript
case "TOGGLE_BODY_PART_DETAIL": {
  const { part, detail } = action.payload
  return {
    ...state,
    bodyParts: state.bodyParts.map((bp) => {
      if (bp.part !== part) return bp
      const current = bp.details ?? []
      return {
        ...bp,
        details: current.includes(detail)
          ? current.filter((d) => d !== detail)
          : [...current, detail],
      }
    }),
  }
}
```

- [ ] **Step 1: 컴포넌트 구현**

```typescript
// src/features/do-workout/ui/BodyPartDetailSheet.tsx
import { Modal, Pressable, ScrollView, Text, View } from "react-native"
import type { BodyPart, BodyPartDetail } from "@/entities/workout-session"
import { BODY_PART_DETAILS } from "@/entities/workout-session"
import { bodyPartDetailLabel, bodyPartLabel } from "@/shared/lib/format"

interface BodyPartDetailSheetProps {
  visible: boolean
  bodyPart: BodyPart | null
  selectedDetails: BodyPartDetail[]
  onToggleDetail: (detail: BodyPartDetail) => void
  onClose: () => void
}

export function BodyPartDetailSheet({
  visible,
  bodyPart,
  selectedDetails,
  onToggleDetail,
  onClose,
}: BodyPartDetailSheetProps) {
  if (!bodyPart) return null
  const details = BODY_PART_DETAILS[bodyPart]

  return (
    <Modal
      visible={visible}
      transparent
      animationType="slide"
      onRequestClose={onClose}
    >
      <Pressable className="flex-1 justify-end" onPress={onClose}>
        <Pressable
          className="bg-[var(--yb-surface-02)] rounded-t-2xl px-6 pt-4 pb-8"
          onPress={() => {}}
        >
          <View className="w-10 h-1 rounded-full bg-[var(--yb-border)] self-center mb-4" />
          <Text className="text-[var(--yb-text-primary)] text-lg font-semibold mb-4">
            {bodyPartLabel(bodyPart)}
          </Text>
          {details.length === 0 ? (
            <Text className="text-[var(--yb-text-secondary)] text-sm">
              세부 부위가 없어요.
            </Text>
          ) : (
            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              contentContainerClassName="flex-row gap-2"
            >
              {details.map((detail) => {
                const active = selectedDetails.includes(detail)
                return (
                  <Pressable
                    key={detail}
                    onPress={() => onToggleDetail(detail)}
                    className={`px-4 py-2 rounded-full border ${
                      active
                        ? "bg-[#9B7E56] border-[#9B7E56]"
                        : "bg-transparent border-[var(--yb-border)]"
                    }`}
                  >
                    <Text
                      className={`text-sm ${
                        active ? "text-white font-medium" : "text-[var(--yb-text-secondary)]"
                      }`}
                    >
                      {bodyPartDetailLabel(detail)}
                    </Text>
                  </Pressable>
                )
              })}
            </ScrollView>
          )}
        </Pressable>
      </Pressable>
    </Modal>
  )
}
```
