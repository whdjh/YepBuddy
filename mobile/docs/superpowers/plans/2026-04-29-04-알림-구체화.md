```typescript
// src/shared/lib/notification/permissions.ts
import * as Notifications from "expo-notifications"
import { Platform } from "react-native"

export async function requestNotificationPermissions(): Promise<boolean> {
  if (Platform.OS === "android") return true

  const { status: existing } = await Notifications.getPermissionsAsync()
  if (existing === "granted") return true

  const { status } = await Notifications.requestPermissionsAsync({
    ios: {
      allowAlert: true,
      allowBadge: true,
      allowSound: true,
      allowProvisional: true,
    },
  })
  return status === "granted"
}
```

```typescript
// src/shared/lib/notification/message.ts
import { getWorkoutSummariesForMonth } from "@/entities/workout-session"
import { getThisWeekDateRange, getLocalDateKey } from "@/shared/lib/date"
import { REMINDER_CATEGORY_ID } from "./categories"

export const REMINDER_TITLE = "YepBuddy"

export function buildReminderBody(weeklyCount: number): string {
  if (weeklyCount === 0) return "이번 주 첫 운동을 시작해볼까요?"
  return `이번 주 ${weeklyCount}번 운동했어요. 오늘도 하러 가볼까요?`
}

export async function buildReminderContent(): Promise<{
  title: string
  body: string
  categoryIdentifier: string
}> {
  const weeklyCount = await getThisWeekHealthKitCount()
  return {
    title: REMINDER_TITLE,
    body: buildReminderBody(weeklyCount),
    categoryIdentifier: REMINDER_CATEGORY_ID,
  }
}

async function getThisWeekHealthKitCount(): Promise<number> {
  try {
    const now = new Date()
    const { startDateKey, endDateKey } = getThisWeekDateRange()
    const summaries = await getWorkoutSummariesForMonth(
      now.getFullYear(),
      now.getMonth() + 1
    )
    return summaries.filter((s) => {
      const dateKey = getLocalDateKey(new Date(s.startDate))
      return dateKey >= startDateKey && dateKey <= endDateKey
    }).length
  } catch {
    return 0
  }
}
```

```typescript
// src/shared/lib/notification/scheduler.ts
import * as Notifications from "expo-notifications"
import AsyncStorage from "@react-native-async-storage/async-storage"
import { buildReminderContent } from "./message"

const REMINDER_ID_KEY = "yb:notification:reminder-id"

export async function getScheduledReminderId(): Promise<string | null> {
  return AsyncStorage.getItem(REMINDER_ID_KEY)
}

export async function saveScheduledReminderId(id: string): Promise<void> {
  await AsyncStorage.setItem(REMINDER_ID_KEY, id)
}

export async function clearScheduledReminderId(): Promise<void> {
  await AsyncStorage.removeItem(REMINDER_ID_KEY)
}

export async function cancelScheduledReminders(): Promise<void> {
  const id = await getScheduledReminderId()
  if (id) {
    await Notifications.cancelScheduledNotificationAsync(id).catch(() => {})
    await clearScheduledReminderId()
  }
}

export async function scheduleReminder22h(completedAtIso: string): Promise<void> {
  await cancelScheduledReminders()

  const content = await buildReminderContent()
  const triggerDate = new Date(
    new Date(completedAtIso).getTime() + 22 * 60 * 60 * 1000
  )

  const id = await Notifications.scheduleNotificationAsync({
    content: {
      title: content.title,
      body: content.body,
      categoryIdentifier: content.categoryIdentifier,
    },
    trigger: {
      type: Notifications.SchedulableTriggerInputTypes.DATE,
      date: triggerDate,
    },
  })
  await saveScheduledReminderId(id)
}

export async function rescheduleFromNow(): Promise<void> {
  await scheduleReminder22h(new Date().toISOString())
}
```

- [ ] **Step 1: 구현**

"운동할게" → 앱 포어그라운드 + `/workout` 이동. "오늘은 쉴래" → 22h 재예약. 알림 탭(기본 액션)도 "운동할게"와 동일 처리한다:

```typescript
// src/shared/lib/notification/handler.ts
import * as Notifications from "expo-notifications"
import { router } from "expo-router"
import { ACTION_DO_WORKOUT, ACTION_SKIP_TODAY } from "./categories"
import { rescheduleFromNow } from "./scheduler"

export function setupNotificationHandler(): () => void {
  const sub = Notifications.addNotificationResponseReceivedListener((response) => {
    const actionId = response.actionIdentifier

    if (
      actionId === ACTION_DO_WORKOUT ||
      actionId === Notifications.DEFAULT_ACTION_IDENTIFIER
    ) {
      router.push("/workout")
      return
    }

    if (actionId === ACTION_SKIP_TODAY) {
      rescheduleFromNow()
      return
    }
  })

  return () => sub.remove()
}
```
