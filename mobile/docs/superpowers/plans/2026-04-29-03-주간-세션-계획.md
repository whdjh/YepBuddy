```typescript
// src/entities/workout-session/model/weeklyPlanStorage.ts
import AsyncStorage from "@react-native-async-storage/async-storage"

export interface WeeklyPlanSettings {
  weeklyTargetSessions: number
}

const KEY = "yb:workout:weekly-plan"

export async function saveWeeklyPlanSettings(s: WeeklyPlanSettings): Promise<void> {
  await AsyncStorage.setItem(KEY, JSON.stringify(s))
}

export async function loadWeeklyPlanSettings(): Promise<WeeklyPlanSettings | null> {
  const raw = await AsyncStorage.getItem(KEY)
  return raw ? (JSON.parse(raw) as WeeklyPlanSettings) : null
}
```

```typescript
// src/features/view-summary/model/useWeeklySessionPlan.ts
import { useCallback, useEffect, useState } from "react"
import {
  loadWeeklyPlanSettings,
  saveWeeklyPlanSettings,
  getStoredWorkoutSessionsInRange,
} from "@/entities/workout-session"
import { getThisWeekDateRange } from "@/shared/lib/date"
import { useTranslation } from "react-i18next"
import type { StoredWorkoutSession, WeeklyPlanSettings } from "@/entities/workout-session"

export interface WeeklySessionSlot {
  index: number
  session: StoredWorkoutSession | null
}

export interface WeeklySessionPlan {
  targetSessions: number
  completedSessions: number
  remainingSessions: number
  slots: WeeklySessionSlot[]
  summaryText: string
  remainingText: string
}

export function useWeeklySessionPlan() {
  const { t } = useTranslation()
  const [settings, setSettings] = useState<WeeklyPlanSettings | null>(null)
  const [sessions, setSessions] = useState<StoredWorkoutSession[]>([])
  const [isLoading, setIsLoading] = useState(true)

  const load = useCallback(async () => {
    setIsLoading(true)
    const loadedSettings = await loadWeeklyPlanSettings()
    const { startDateKey, endDateKey } = getThisWeekDateRange()
    const weekSessions = loadedSettings
      ? await getStoredWorkoutSessionsInRange(startDateKey, endDateKey)
      : []
    setSettings(loadedSettings)
    setSessions(weekSessions)
    setIsLoading(false)
  }, [])

  useEffect(() => { load() }, [load])

  const updateTargetSessions = useCallback(async (target: number) => {
    const next: WeeklyPlanSettings = { weeklyTargetSessions: target }
    await saveWeeklyPlanSettings(next)
    setSettings(next)
  }, [])

  if (!settings) {
    return { plan: null, settings: null, isLoading, updateTargetSessions, reload: load }
  }

  const target = settings.weeklyTargetSessions
  const completed = sessions.length
  const remaining = Math.max(0, target - completed)

  const plan: WeeklySessionPlan = {
    targetSessions: target,
    completedSessions: completed,
    remainingSessions: remaining,
    slots: Array.from({ length: target }, (_, i) => ({
      index: i,
      session: sessions[i] ?? null,
    })),
    summaryText: t("workout.weekly.summary", { target, completed }),
    remainingText: t("workout.weekly.remaining", { remaining }),
  }

  return { plan, settings, isLoading, updateTargetSessions, reload: load }
}
```

```typescript
// src/features/view-summary/ui/WeeklySessionCard.tsx
import { View, Text } from "react-native"
import type { StoredWorkoutSession } from "@/entities/workout-session"
import { bodyPartLabel } from "@/shared/lib/format"

interface WeeklySessionCardProps {
  session: StoredWorkoutSession | null
  index: number
}

export function WeeklySessionCard({ session, index }: WeeklySessionCardProps) {
  if (!session) {
    return (
      <View className="w-12 h-16 rounded-xl border border-dashed border-[var(--yb-border)] items-center justify-center">
        <Text className="text-[var(--yb-text-tertiary)] text-xs">{index + 1}</Text>
      </View>
    )
  }

  const primary = session.bodyParts.reduce(
    (max, bp) => (bp.setCount > max.setCount ? bp : max),
    session.bodyParts[0],
  )

  return (
    <View className="w-12 h-16 rounded-xl bg-[#9B7E56] items-center justify-center gap-1">
      <Text className="text-white text-xs font-bold">{index + 1}</Text>
      {primary && (
        <Text className="text-white text-[9px] text-center px-1" numberOfLines={1}>
          {bodyPartLabel(primary.part)}
        </Text>
      )}
    </View>
  )
}
```

```typescript
// src/features/view-summary/ui/WeeklySessionList.tsx
import { Pressable, Text, View } from "react-native"
import { useTranslation } from "react-i18next"
import { useWeeklySessionPlan } from "../model/useWeeklySessionPlan"
import { WeeklySessionCard } from "./WeeklySessionCard"

interface WeeklySessionListProps {
  onPressSettings?: () => void
}

export function WeeklySessionList({ onPressSettings }: WeeklySessionListProps) {
  const { t } = useTranslation()
  const { plan, settings, isLoading } = useWeeklySessionPlan()

  if (isLoading) return null

  if (!settings) {
    return (
      <Pressable
        onPress={onPressSettings}
        className="py-3 px-4 rounded-xl border border-dashed border-[var(--yb-border)] items-center"
      >
        <Text className="text-[var(--yb-text-secondary)] text-sm">
          {t("workout.weekly.settings.placeholder")}
        </Text>
      </Pressable>
    )
  }

  return (
    <View className="gap-3">
      <View className="flex-row justify-between items-center">
        <Text className="text-[var(--yb-text-primary)] text-sm font-medium">
          {plan?.summaryText}
        </Text>
        <Pressable onPress={onPressSettings}>
          <Text className="text-[var(--yb-text-tertiary)] text-xs">
            {t("workout.weekly.settings.title")}
          </Text>
        </Pressable>
      </View>
      <View className="flex-row gap-2 flex-wrap">
        {plan?.slots.map((slot) => (
          <WeeklySessionCard key={slot.index} session={slot.session} index={slot.index} />
        ))}
      </View>
      <Text className="text-[var(--yb-text-secondary)] text-xs">{plan?.remainingText}</Text>
    </View>
  )
}
```

```typescript
// src/features/view-summary/ui/WeeklyPlanSettingsSheet.tsx
import { useState } from "react"
import { Modal, Pressable, Text, View } from "react-native"
import { useTranslation } from "react-i18next"
import { useWeeklySessionPlan } from "../model/useWeeklySessionPlan"

interface WeeklyPlanSettingsSheetProps {
  visible: boolean
  onClose: () => void
}

export function WeeklyPlanSettingsSheet({ visible, onClose }: WeeklyPlanSettingsSheetProps) {
  const { t } = useTranslation()
  const { settings, updateTargetSessions } = useWeeklySessionPlan()
  const [selected, setSelected] = useState(settings?.weeklyTargetSessions ?? 5)

  const handleSave = async () => {
    await updateTargetSessions(selected)
    onClose()
  }

  return (
    <Modal visible={visible} transparent animationType="slide" onRequestClose={onClose}>
      <Pressable className="flex-1 justify-end" onPress={onClose}>
        <Pressable
          className="bg-[var(--yb-surface-02)] rounded-t-2xl px-6 pt-4 pb-10"
          onPress={() => {}}
        >
          <View className="w-10 h-1 rounded-full bg-[var(--yb-border)] self-center mb-6" />
          <Text className="text-[var(--yb-text-primary)] text-lg font-semibold mb-6">
            {t("workout.weekly.settings.title")}
          </Text>
          <View className="flex-row justify-center gap-2 mb-8">
            {[1, 2, 3, 4, 5, 6, 7].map((n) => (
              <Pressable
                key={n}
                onPress={() => setSelected(n)}
                className={`w-10 h-10 rounded-full items-center justify-center border ${
                  selected === n
                    ? "bg-[#9B7E56] border-[#9B7E56]"
                    : "border-[var(--yb-border)]"
                }`}
              >
                <Text
                  className={`text-sm font-medium ${
                    selected === n ? "text-white" : "text-[var(--yb-text-secondary)]"
                  }`}
                >
                  {n}
                </Text>
              </Pressable>
            ))}
          </View>
          <Pressable
            onPress={handleSave}
            className="bg-[#9B7E56] rounded-xl py-3 items-center"
          >
            <Text className="text-white font-semibold">{t("workout.weekly.settings.save")}</Text>
          </Pressable>
        </Pressable>
      </Pressable>
    </Modal>
  )
}
```
