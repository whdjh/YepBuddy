```typescript
// src/features/share-workout/lib/pickWorkoutPhoto.ts
import * as ImagePicker from "expo-image-picker"

export async function pickWorkoutPhoto(): Promise<string | null> {
  const { status } = await ImagePicker.requestMediaLibraryPermissionsAsync()
  if (status !== "granted") return null

  const result = await ImagePicker.launchImageLibraryAsync({
    mediaTypes: ["images"],
    allowsEditing: true,
    aspect: [4, 5],
    quality: 0.8,
  })

  if (result.canceled) return null
  return result.assets[0].uri
}
```

- [ ] **Step 2: saveWorkoutImage.ts**

```typescript
// src/features/share-workout/lib/saveWorkoutImage.ts
import * as MediaLibrary from "expo-media-library"

export async function saveWorkoutImage(uri: string): Promise<boolean> {
  const { status } = await MediaLibrary.requestPermissionsAsync()
  if (status !== "granted") return false
  try {
    await MediaLibrary.saveToLibraryAsync(uri)
    return true
  } catch {
    return false
  }
}
```

- [ ] **Step 3: shareWorkoutImage.ts**

```typescript
// src/features/share-workout/lib/shareWorkoutImage.ts
import * as Sharing from "expo-sharing"
import { Share } from "react-native"

export async function shareWorkoutImage(uri: string): Promise<void> {
  const available = await Sharing.isAvailableAsync()
  if (available) {
    await Sharing.shareAsync(uri, { mimeType: "image/png" })
  } else {
    await Share.share({ url: uri })
  }
}
```

```typescript
// src/features/share-workout/ui/WorkoutShareCard.tsx
import { forwardRef } from "react"
import { Image, Text, View } from "react-native"
import type { StoredWorkoutSession } from "@/entities/workout-session"
import { bodyPartLabel, formatDuration } from "@/shared/lib/format"

interface WorkoutShareCardProps {
  session: StoredWorkoutSession
  backgroundImageUri?: string
}

export const WorkoutShareCard = forwardRef<View, WorkoutShareCardProps>(
  ({ session, backgroundImageUri }, ref) => {
    const durationSec = Math.floor(
      (new Date(session.completedAt).getTime() -
        new Date(session.startedAt).getTime()) /
        1000
    )
    const dateStr = new Date(session.startedAt).toLocaleDateString("ko-KR", {
      year: "numeric",
      month: "long",
      day: "numeric",
    })
    const summary = session.bodyParts
      .map((bp) => `${bodyPartLabel(bp.part)} ${bp.setCount}세트`)
      .join(" · ")

    return (
      <View
        ref={ref}
        style={{ width: 320, aspectRatio: 4 / 5 }}
        className="rounded-2xl overflow-hidden bg-black"
        collapsable={false}
      >
        {backgroundImageUri ? (
          <Image
            source={{ uri: backgroundImageUri }}
            className="absolute inset-0 w-full h-full"
            resizeMode="cover"
          />
        ) : (
          <View className="absolute inset-0 bg-[#1A1A1A]" />
        )}
        <View className="absolute inset-0 bg-black/50" />
        <View className="absolute bottom-0 left-0 right-0 p-5 gap-2">
          <Text className="text-[#9B7E56] text-xs font-semibold tracking-widest">
            YEPBUDDY
          </Text>
          <Text className="text-white text-xl font-bold">{summary}</Text>
          <Text className="text-white/70 text-sm">{formatDuration(durationSec)}</Text>
          <Text className="text-white/40 text-xs">{dateStr}</Text>
        </View>
      </View>
    )
  }
)

WorkoutShareCard.displayName = "WorkoutShareCard"
```

- [ ] **Step 3: 컴포넌트 내부에 상태와 핸들러 추가**

```typescript
const cardRef = useRef<ViewShot>(null)
const [bgUri, setBgUri] = useState<string | undefined>(undefined)
const [capturing, setCapturing] = useState(false)

const handlePickPhoto = async () => {
  const uri = await pickWorkoutPhoto()
  if (uri) setBgUri(uri)
}

const handleShare = async () => {
  setCapturing(true)
  try {
    const uri = await cardRef.current?.capture?.()
    if (!uri) return
    Alert.alert("오운완 공유", "", [
      {
        text: "사진 앱에 저장",
        onPress: async () => {
          const ok = await saveWorkoutImage(uri)
          if (ok) Alert.alert("저장됐어요!")
        },
      },
      { text: "공유하기", onPress: () => shareWorkoutImage(uri) },
      { text: "취소", style: "cancel" },
    ])
  } finally {
    setCapturing(false)
  }
}
```
